package compatconsumer

import (
	"encoding/binary"
	"os"
	"testing"

	x "compatconsumer/extracted"
)

func TestLandAgainstNativeLean(t *testing.T) {
	oracle, err := os.ReadFile("oracle.bin")
	if err != nil {
		t.Fatal(err)
	}
	const words = 1 << 16
	if len(oracle) != 6*words {
		t.Fatalf("native output has %d bytes, want %d", len(oracle), 6*words)
	}
	limits := x.DefaultLimits()
	for i := range words {
		a := uint16(i)
		for j, b := range [...]uint16{0, 0xffff, uint16(uint32(i)*40503 + 7919)} {
			native := binary.BigEndian.Uint16(oracle[6*i+2*j:])
			got, err := x.RunTyped(a, b, limits)
			if err != nil || got != native || got != a&b {
				t.Fatalf("%04x & %04x: generated %04x, native %04x, Go %04x, err %v", a, b, got, native, a&b, err)
			}
		}
	}
}
