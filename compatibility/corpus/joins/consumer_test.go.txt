package compatconsumer

import (
	x "compatconsumer/extracted"
	"testing"
)

func TestJoins(t *testing.T) {
	for _, tc := range []struct {
		a, b                bool
		x, y, z, bias, want uint32
	}{
		{true, true, 2, 5, 9, 3, 14}, {false, true, 2, 5, 9, 3, 17},
		{true, false, 2, 5, 9, 3, 21}, {false, false, 2, 5, 4294967295, 1, 3},
	} {
		out, err := x.RunTyped(tc.a, tc.b, tc.x, tc.y, tc.z, tc.bias, x.DefaultLimits())
		if err != nil || out != tc.want {
			t.Fatal(tc, out, err)
		}
	}
}
